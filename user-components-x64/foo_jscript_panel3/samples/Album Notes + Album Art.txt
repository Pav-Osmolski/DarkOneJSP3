var margin = _scale(12);

var panel = new _panel({ enhanced_page_background : true });
var album_notes = new _album_notes(0, 0, 0, 0);
var albumart_appearance = new _combined_artwork('2K3.ALBUM.NOTES.ART', 'Display album art', {
	layout_changed : function () { on_size(); },
	background_changed : function () { albumart.metadb_changed(); },
});
var albumart = new _albumart(margin, margin, 0, 0, {
	review_panel : true,
	appearance : albumart_appearance,
});
var album_notes_paint = _.bind(album_notes.paint, album_notes);

album_notes.paint = function (gr) {
	albumart.paint_background(gr);

	var y = !albumart_appearance.displayed()
		? 0
		: albumart.properties.layout.value == 0 ? margin : this.y - margin;
	gr.WriteTextSimple(this.header_text(), panel.fonts.title, panel.colours.highlight, this.x, y, this.w, TM, DWRITE_TEXT_ALIGNMENT_LEADING, DWRITE_PARAGRAPH_ALIGNMENT_CENTER, DWRITE_WORD_WRAPPING_NO_WRAP, DWRITE_TRIMMING_GRANULARITY_CHARACTER);
	album_notes_paint(gr);
}

var album_notes_menu = {
	rbtn_up : function (x, y) {
		albumart_appearance.append_menu(panel.m);
		album_notes.rbtn_up(x, y);
	},
	rbtn_up_done : function (idx) {
		if (!albumart_appearance.handle_menu(idx))
			album_notes.rbtn_up_done(idx);
	},
};

panel.item_focus_change();

function on_colours_changed() {
	panel.colours_changed();
	window.Repaint();
}

function on_font_changed() {
	panel.font_changed();
	window.Repaint();
}

function on_http_request_done(task_id, success, response_text, status, response_headers) {
	album_notes.http_request_done(task_id, success, response_text, status, response_headers);
}

function on_item_focus_change() {
	if (panel.prefer_playing())
		return;

	panel.item_focus_change();
}

function on_key_down(k) {
	album_notes.key_down(k);
	albumart.key_down(k);
}

function on_metadb_changed(handles, fromhook) {
	if (fromhook)
		return;

	album_notes.metadb_changed();
	albumart.metadb_changed();
}

function on_mouse_lbtn_up(x, y) {
	album_notes.lbtn_up(x, y);
}

function on_mouse_move(x, y) {
	album_notes.move(x, y);
	albumart.move(x, y);
}

function on_mouse_leave() {
	album_notes.move(-1, -1);
	albumart.move(-1, -1);
	_tt('');
}

function on_mouse_lbtn_dblclk(x, y) {
	albumart.lbtn_dblclk(x, y);
}

function on_notify_data(name, info) {
	if (jsp3EnhancedHandleSampleReset(name, info, ["album-notes", "musicbrainz"])) return;
	album_notes.notify_data(name, info);
}

function on_mouse_rbtn_up(x, y) {
	if (albumart_appearance.displayed() && albumart.containsXY(x, y))
		return panel.rbtn_up(x, y, albumart);

	return panel.rbtn_up(x, y, album_notes_menu);
}

function on_mouse_wheel(s) {
	if (albumart_appearance.displayed() && utils.IsKeyPressed(VK_CONTROL)) {
		var value = _clamp(albumart.properties.ratio.value - (s * 0.05), 0.2, 0.8);
		if (value != albumart.properties.ratio.value) {
			albumart.properties.ratio.value = value;
			on_size();
			window.Repaint();
		}
	} else {
		albumart.wheel(s);
		album_notes.wheel(s);
	}
}

function on_paint(gr) {
	panel.paint(gr);
	album_notes.paint(gr);
	albumart.paint(gr);
}

function on_playback_dynamic_info_track(type) {
	panel.playback_colours_changed(type);
	if (type == 0) {
		window.Repaint();
	} else {
		albumart.metadb_changed();
	}
}

function on_playback_new_track() {
	panel.playback_colours_changed();
	panel.item_focus_change();
}

function on_playback_stop(reason) {
	if (reason != 2) {
		panel.playback_colours_changed();
		panel.item_focus_change();
	}
}

function on_playlist_switch() {
	on_item_focus_change();
}

function on_script_unload() {
	albumart.dispose();
	album_notes.dispose();
}

function on_size() {
	panel.size();
	if (!albumart_appearance.displayed()) {
		albumart.x = albumart.y = albumart.w = albumart.h = 0;
		album_notes.x = LM;
		album_notes.y = TM;
		album_notes.w = Math.max(1, panel.w - (LM * 2));
		album_notes.h = Math.max(_scale(24), panel.h - TM);
		album_notes.size();
		return;
	}

	albumart.x = margin;
	albumart.y = margin;
	var available_w = Math.max(1, panel.w - (margin * 3));
	var available_h = Math.max(1, panel.h - (margin * 3));
	var minimum_text_h = _scale(24);

	switch (albumart.properties.layout.value) {
	case 0: // Album Art left, Text right
		albumart.w = Math.max(1, Math.min(Math.round(panel.w * albumart.properties.ratio.value), available_w));
		albumart.h = Math.max(1, panel.h - (margin * 2));

		album_notes.x = albumart.w + (margin * 2);
		album_notes.y = margin * 2;
		album_notes.w = Math.max(1, panel.w - albumart.w - (margin * 3));
		album_notes.h = Math.max(minimum_text_h, panel.h - album_notes.y);
		break;
	case 1: // Album Art top, Text bottom
		albumart.w = Math.max(1, panel.w - (margin * 2));
		albumart.h = Math.max(1, Math.min(Math.round(panel.h * albumart.properties.ratio.value), available_h));

		album_notes.x = margin;
		album_notes.y = albumart.h + (margin * 3);
		album_notes.w = Math.max(1, panel.w - (margin * 2));
		album_notes.h = Math.max(minimum_text_h, panel.h - albumart.h - (margin * 3));
		break;
	}

	album_notes.size();
}
