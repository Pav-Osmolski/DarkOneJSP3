// row-spacing, row-aware scroll-state and optional alternating-row shading.

g_playlist_manager_cadence_reporter.start();

function on_notify_data(name, info) {
    if (g_playlist_manager_cadence_reporter && g_playlist_manager_cadence_reporter.handleNotification(name, info)) return;
    jsp3EnhancedHandleSampleReset(name, info, "playlist-manager");
}
